~~~python
def greet_user(name): print(f"Здравствуйте, {name}!")
def simple_calculator(number1, number2, operator):
    return {"+": number1 + number2, "-": number1 - number2, "*": number1 * number2, "/": number1 / number2}[operator]
def power(number, degree=2): return number ** degree
def find_max(number1, number2): return max(number1, number2)
def display_user_profile(name, age, city): print(name, age, city)
def is_palindrome(text):
    text = "".join(text.lower().split())
    return text == text[::-1]

greet_user("Денис")
print(simple_calculator(8, 2, "+"))
print(power(5), power(2, 3), find_max(7, 4))
display_user_profile(city="Казань", name="Денис", age=20)
print(is_palindrome("А роза упала на лапу Азора"))
~~~
