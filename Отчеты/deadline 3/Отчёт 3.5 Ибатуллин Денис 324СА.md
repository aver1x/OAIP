# 1

def find_item(items, target):  # O(n)
    for item in items:
        if item == target: return True
    return False

def first_last(items):  # O(1)
    return items[0], items[-1]

def nested(items):  # O(n^2)
    return [(left, right) for left in items for right in items]

def find_pair_fast(values, target):
    seen = set()
    for value in values:
        if target - value in seen: return target - value, value
        seen.add(value)
    return None

target = -1
print(target in values)  # линейный поиск O(n)
left, right = 0, len(values) - 1
while left <= right:
    middle = (left + right) // 2
    if values[middle] == target: break
    if values[middle] < target: left = middle + 1
    else: right = middle - 1
