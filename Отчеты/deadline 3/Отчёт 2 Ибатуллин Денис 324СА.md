~~~python
def squares_gen(n):
    for number in range(1, n + 1): yield number ** 2

even_numbers = (number for number in range(10, 21) if number % 2 == 0)

def fibonacci(n):
    first, second = 0, 1
    for _ in range(n):
        yield first
        first, second = second, first + second

def my_range(start, end, step):
    while (step > 0 and start < end) or (step < 0 and start > end):
        yield start
        start += step

def filter_logs(logs, keyword):
    for log in logs:
        if keyword in log: yield log

def flat_gen(nested_list):
    for item in nested_list: yield from item

print(list(squares_gen(5)))
print(list(even_numbers))
print(list(fibonacci(8)))
print(list(my_range(0, 1, 0.2)))
print(list(filter_logs(["INFO start", "ERROR fail"], "ERROR")))
print(list(flat_gen([[1, 2], [3], [4, 5]])))
~~~
